server.js (Main backend file)
import express from 'express';
import mongoose from 'mongoose';
import cors from 'cors';
import dotenv from 'dotenv';
import bookRoutes from './routes/bookRoutes.js';

dotenv.config();
const app = express();

app.use(cors());
app.use(express.json());
app.use('/api/books', bookRoutes);

mongoose.connect(process.env.MONGO_URI, {
  useNewUrlParser: true,
  useUnifiedTopology: true,
})
.then(() => console.log('✅ MongoDB connected'))
.catch(err => console.error('❌ Mongo Error:', err));

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => {
  console.log(`🚀 Server running at http://localhost:${PORT}`);
});


📦 4. Mongoose Model
📄 backend/models/Book.js

import mongoose from 'mongoose';

const bookSchema = new mongoose.Schema({
  title: String,
  author: String,
  available: { type: Boolean, default: true },
  bookedBy: { type: String, default: '' }
});

const Book = mongoose.model('Book', bookSchema);
export default Book;


🔁 5. Routes
📄 backend/routes/bookRoutes.js

import express from 'express';
import Book from '../models/Book.js';

const router = express.Router();

router.get('/', async (req, res) => {
  const books = await Book.find();
  res.json(books);
});

router.post('/', async (req, res) => {
  const { title, author } = req.body;
  const book = new Book({ title, author });
  await book.save();
  res.status(201).json(book);
});

router.put('/:id/book', async (req, res) => {
  const { id } = req.params;
  const { bookedBy } = req.body;
  const book = await Book.findByIdAndUpdate(id, {
    available: false,
    bookedBy
  }, { new: true });
  res.json(book);
});

export default router;

Step 2: Import the CSS into Your React Application
Make sure to import your index.css in src/main.jsx (or src/index.jsx if you're using JSX) to apply the styles globally.
// src/index.js or src/index.jsx
import React from "react";
import ReactDOM from "react-dom/client";
import "./styles.css"; // Importing your custom CSS
import App from "./App";

ReactDOM.createRoot(document.getElementById("root")).render(
  <React.StrictMode>
    <App />
  </React.StrictMode>
);

This will apply the styles defined in styles.css to the entire app.
